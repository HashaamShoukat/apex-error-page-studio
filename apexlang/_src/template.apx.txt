            /* Modern Error: the Error Page Studio designer previews a look through the URL hash (#me=me-Design--neon ...) */
            window.modernError = (function (body) {
              var LOOK = /^me-(Design|Art|Accent|Scheme|Bg|Width|Anim|Fx)-/;
              function applyPreview() {
                var m = /[#&]me=([^&]*)/.exec(window.location.hash);
                if (!m) { return; }
                Array.prototype.slice.call(body.classList).forEach(function (c) { if (LOOK.test(c)) { body.classList.remove(c); } });
                decodeURIComponent(m[1]).split(/[\s,]+/).forEach(function (c) { if (LOOK.test(c)) { body.classList.add(c); } });
              }
              try { applyPreview(); } catch (e) { /* never block an error page */ }
              return { applyPreview: applyPreview };
            })(document.body);
            </script>

    javaScript {
        executeWhenPageLoads:
            ```javascript-browser
            // Modern Error behaviour, switched by template options; texts from text messages MODERN_ERROR.* when defined
            (function ($, me) {
                var body = document.body;
                function has(c) { return body.classList.contains(c); }
                function msg(key, fallback) {
                    var t = apex.lang.getMessage(key);
                    return (!t || t === key) ? fallback : t;
                }

                // designer preview: follow hash changes without a reload
                window.addEventListener("hashchange", function () { me.applyPreview(); });

                // Back: the previous page, or the home page when there is none
                $(document).on("click", ".me-Back", function (e) {
                    e.preventDefault();
                    if (window.history.length > 1) { window.history.back(); } else { window.location.href = $(".me-Home").attr("href") || "./"; }
                });

                // Try Again
                $(document).on("click", ".me-Reload", function (e) { e.preventDefault(); window.location.reload(); });

                // Copy Error Details (APEX error page): message, details, page and time - ready for a support ticket
                var err = $(".me-Error");
                if (err.length && has("me-Fx--copy") && navigator.clipboard) {
                    $('<button type="button" class="me-Btn me-Copy"></button>')
                        .text(msg("MODERN_ERROR.COPY", "Copy details"))
                        .appendTo(err.find(".me-Actions"))
                        .on("click", function () {
                            var b = this,
                                text = [err.find(".me-Title").text(), err.find(".me-Text").text(), err.find(".me-Tech").text(),
                                        "Page: " + window.location.href.split("#")[0], "Time: " + new Date().toISOString()]
                                       .map(function (s) { return $.trim(s); }).filter(Boolean).join("\n");
                            navigator.clipboard.writeText(text).then(function () {
                                b.classList.add("me-Copied");
                                setTimeout(function () { b.classList.remove("me-Copied"); }, 2000);
                            });
                        });
                }

                // Reload When Back Online: an offline page comes back by itself
                if (has("me-Fx--online")) {
                    window.addEventListener("online", function () { window.location.reload(); });
                }
            })(apex.jQuery, window.modernError);
            ```
